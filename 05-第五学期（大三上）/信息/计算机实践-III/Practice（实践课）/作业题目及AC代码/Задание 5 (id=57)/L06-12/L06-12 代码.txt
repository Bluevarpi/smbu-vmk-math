#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <fcntl.h>

struct word_item {
	char *word;
	struct word_item *next;
};

static void free_list(struct word_item *head) {
    struct word_item *p = head, *q;
    while (p) q = p->next, free(p->word), free(p), p = q;
}

static int add_word(struct word_item **h, struct word_item **t, const char *w) {
    struct word_item *n;
    if (!(n = malloc(sizeof(*n))) || !(n->word = malloc(strlen(w) + 1))) return free(n), -1;
    return strcpy(n->word, w), n->next = NULL, *t = *h ? ((*t)->next = n) : (*h = n), 0;
}

static char **list_to_array(struct word_item *h, int *cnt) {
    int n = 0;
    struct word_item *p = h;
    while(p) n++, p = p->next;
    char **arr = malloc(n * sizeof(*arr));
    if (!arr) return *cnt = 0, NULL;
    for (*cnt = n, n = 0, p = h; p; p = p->next) arr[n++] = p->word;
    return arr;
}

static int parse_command(char **tokens, int start, int end, char ***argv_out, char **in_file_out, char **out_file_out, int *out_mode_out) {
    char *in_file = NULL, *out_file = NULL, **argv;
	int out_mode = 0, j, wrong = 0, argc = 0, k;
    for (j = start; j < end; j++) {
        if (!tokens[j]) continue;
        if (strcmp(tokens[j], "<") == 0) {
            if ((in_file || j + 1 >= end || !tokens[j + 1]) && (wrong = 1)) break;
            in_file = tokens[j + 1], tokens[j] = tokens[j + 1] = NULL, j++;
			continue;
        }
        if (strcmp(tokens[j], ">") == 0) {
            if ((out_file || j + 1 >= end || !tokens[j + 1]) && (wrong = 1)) break;
            out_mode = 1, out_file = tokens[j + 1], tokens[j] = tokens[j + 1] = NULL, j++;
			continue;
        }
        if (strcmp(tokens[j], ">>") == 0) {
            if ((out_mode || j + 1 >= end || !tokens[j + 1]) && (wrong = 1)) break;
            out_mode = 2; out_file = tokens[j + 1]; tokens[j] = tokens[j + 1] = NULL, j++; continue;
        }
    }
    if (wrong || ({ for (j = start; j < end; j++) if (tokens[j]) argc++; !argc; }) || !(argv = malloc((argc + 1) * sizeof(*argv)))) return -1;
    for (k = 0, j = start; j < end; j++) if (tokens[j]) argv[k++] = tokens[j];
    return argv[k] = NULL, *argv_out = argv, *in_file_out = in_file, *out_file_out = out_file, *out_mode_out = out_mode, 0;
}

int main(void) {
    char buf[1024];
    while (fgets(buf, sizeof(buf), stdin)) {
        struct word_item *head = NULL, *tail = NULL;
        char *cur = NULL, c;
		size_t cur_len = 0, cur_cap = 0;
        int in_quotes = 0, quotes_count = 0, error = 0, i, len = strlen(buf);
        if (len && buf[len - 1] == '\n') buf[len - 1] = '\0';
        for (i = 0; (c = buf[i]) != '\0'; ++i) {
            if (c == '\"' && (in_quotes = !in_quotes, quotes_count++, 1)) continue;
            if (!in_quotes) {
                if (c == ' ' || c == '\t') {
                    if (cur_len && (add_word(&head, &tail, cur) ? (error = 1) : (cur_len = 0, 0))) break;
                    continue;
                }
                if (c == '<') {
                    if (cur_len && (add_word(&head, &tail, cur) ? (error = 1) : (cur_len = 0, 0))) break;
                    if (add_word(&head, &tail, "<") && (error = 1)) break;
                    continue;
                }
                if (c == '>') {
                    if (cur_len && (add_word(&head, &tail, cur) ? (error = 1) : (cur_len = 0, 0))) break;
                    if (buf[i + 1] == '>') {
						if (add_word(&head, &tail, ">>") && (error = 1)) break;
						i++;
					}
                    else if (add_word(&head, &tail, ">") && (error = 1)) break;
                    continue;
                }
                if (c == '|') {
                    if (cur_len && (add_word(&head, &tail, cur) ? (error = 1) : (cur_len = 0, 0))) break;
                    if (add_word(&head, &tail, "|") && (error = 1)) break;
                    continue;
                }
            }
            if (cur_len + 1 >= cur_cap) {
                size_t new_cap = cur_cap ? cur_cap * 2 : 16;
				char *tmp = (char *)realloc(cur, new_cap);
                if (!tmp && (error = 1)) break;
                cur = tmp, cur_cap = new_cap;
            }
            cur[cur_len++] = c, cur[cur_len] = '\0';
        }
        if (!error && cur_len) if (add_word(&head, &tail, cur)) error = 1;
        free(cur);
        if (error) return free_list(head), 1;
        if ((quotes_count & 1) && (printf("Error: unmatched quotes\n"), fflush(stdout), free_list(head), 1)) continue;
		if (!head) continue;
		int tokc, j, pipe_count = 0, *pipe_pos = NULL, cmd_count, wrong = 0, seg;
    	char **tokens = list_to_array(head, &tokc);
        if (!tokens && tokc > 0) return free_list(head), 1;
        for (j = 0; j < tokc; j++) if (tokens[j] && strcmp(tokens[j], "|") == 0) pipe_count++;
        cmd_count = pipe_count + 1;
        if (pipe_count > 0) {
            int idx = 0;
            pipe_pos = (int *)malloc(pipe_count * sizeof(*pipe_pos));
            if (!pipe_pos) return free(tokens), free_list(head), 1;
            for (j = 0; j < tokc; j++) if (tokens[j] && strcmp(tokens[j], "|") == 0) pipe_pos[idx++] = j;
        }
        char ***argvv = (char ***)malloc(cmd_count * sizeof(*argvv)), **in_files = (char **)malloc(cmd_count * sizeof(*in_files)), **out_files = (char **)malloc(cmd_count * sizeof(*out_files));
        int *out_modes = (int *)malloc(cmd_count * sizeof(*out_modes));
        if (!argvv || !in_files || !out_files || !out_modes) return free(argvv), free(in_files), free(out_files), free(out_modes), free(pipe_pos), free(tokens), free_list(head), 1;
        for (seg = 0; seg < cmd_count; seg++) argvv[seg] = NULL, in_files[seg] = NULL, out_files[seg] = NULL, out_modes[seg] = 0;
        for (seg = 0; seg < cmd_count; seg++) {
            int start = (seg == 0) ? 0 : (pipe_pos[seg - 1] + 1);
            int end = (seg == cmd_count - 1) ? tokc : pipe_pos[seg];
            for (j = start; j < end; j++) if (tokens[j] && strcmp(tokens[j], "|") == 0) tokens[j] = NULL;
            if (parse_command(tokens, start, end, &argvv[seg], &in_files[seg], &out_files[seg], &out_modes[seg]) < 0 && (wrong = 1)) break;
        }
        if (!wrong) {
            for (seg = 0; seg < cmd_count; seg++) {
                if (seg > 0 && in_files[seg] && (wrong = 1)) break;
                if (seg < cmd_count - 1 && out_modes[seg] != 0 && (wrong = 1)) break;
            }
        }
        if (wrong) {
            printf("Error: wrong token\n"), fflush(stdout);
            for (seg = 0; seg < cmd_count; seg++) if (argvv[seg]) free(argvv[seg]);
            free(argvv), free(in_files), free(out_files), free(out_modes), free(pipe_pos), free(tokens), free_list(head);
            continue;
        }
        pid_t *pids = (pid_t *)malloc(cmd_count * sizeof(*pids));
        if (!pids) {
            for (seg = 0; seg < cmd_count; seg++) if (argvv[seg]) free(argvv[seg]);
            free(argvv), free(in_files), free(out_files), free(out_modes), free(pipe_pos), free(tokens), free_list(head);
            return 1;
        }
        fflush(NULL);
        int prev_read_fd = -1;
        for (int i_cmd = 0; i_cmd < cmd_count; i_cmd++) {
            int pfd[2], need_pipe = (i_cmd < cmd_count - 1); pid_t pid;
            if (need_pipe) {
                if (pipe(pfd) < 0) {
                    perror("pipe"), free(pids);
                    for (seg = 0; seg < cmd_count; seg++) if (argvv[seg]) free(argvv[seg]);
                    return free(argvv), free(in_files), free(out_files), free(out_modes), free(pipe_pos), free(tokens), free_list(head), 1;
                }
            }
            pid = fork();
            if (pid < 0) {
                perror("fork"), free(pids);
                for (seg = 0; seg < cmd_count; seg++) if (argvv[seg]) free(argvv[seg]);
                return free(argvv), free(in_files), free(out_files), free(out_modes), free(pipe_pos), free(tokens), free_list(head), 1;
            }
            if (pid == 0) {
                if (i_cmd == 0) {
                    if (in_files[0]) {
                        int fd_in = open(in_files[0], O_RDONLY);
                        if (fd_in < 0) perror(in_files[0]), _exit(1);
                        if (dup2(fd_in, STDIN_FILENO) < 0) perror("dup2"), close(fd_in), _exit(1);
                        close(fd_in);
                    }
                } else if (dup2(prev_read_fd, STDIN_FILENO) < 0) perror("dup2"), _exit(1);
                if (i_cmd < cmd_count - 1) {
                    if (dup2(pfd[1], STDOUT_FILENO) < 0) perror("dup2"), _exit(1);
                } else if (out_modes[i_cmd] && out_files[i_cmd]) {
                        int flags = O_WRONLY | O_CREAT | (out_modes[i_cmd] == 1 ? O_TRUNC : O_APPEND);
                        int fd_out = open(out_files[i_cmd], flags, 0666);
                        if (fd_out < 0) perror(out_files[i_cmd]), _exit(1);
                        if (dup2(fd_out, STDOUT_FILENO) < 0) perror("dup2"), close(fd_out), _exit(1);
                        close(fd_out);
                    }
                if (prev_read_fd != -1) close(prev_read_fd);
                if (need_pipe) close(pfd[0]), close(pfd[1]);
                execvp(argvv[i_cmd][0], argvv[i_cmd]), perror(argvv[i_cmd][0]), _exit(1);
            } else {
                pids[i_cmd] = pid;
                if (prev_read_fd != -1) close(prev_read_fd);
                if (need_pipe) prev_read_fd = pfd[0], close(pfd[1]);
            }
        }
        if (prev_read_fd != -1) close(prev_read_fd);
        for (int i_cmd = 0; i_cmd < cmd_count; i_cmd++) waitpid(pids[i_cmd], NULL, 0);
        free(pids);
        for (seg = 0; seg < cmd_count; seg++) if (argvv[seg]) free(argvv[seg]);
        free(argvv), free(in_files), free(out_files), free(out_modes), free(pipe_pos), free(tokens), free_list(head);
    }
    return 0;
}
