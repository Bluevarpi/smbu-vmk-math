#include <stdio.h>
#include <stdbool.h>

bool isPrime (int n) {
	if (n == 2 || n == 3) return true;
	if (n <= 1 || n % 2 == 0 || n % 3 == 0) return false;
	for (int i = 5; i * i <= n; i += 6) if (!(n % i) || !(n % (i + 2))) return false;
	return true;
}

int main () {
	int n;
	scanf("%d", &n);
	printf(isPrime(n) ? "YES" : "NO");
	return 0;
}
